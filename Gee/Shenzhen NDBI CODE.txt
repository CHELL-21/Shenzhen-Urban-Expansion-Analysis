var area = ee.FeatureCollection("projects/practice-1-503311/assets/Shenzhen");

function clippedimg(img){
  return img.clip(area)
};
var s2 = ee.ImageCollection('LANDSAT/LT05/C02/T1_L2')
.filterBounds(area)
.filterDate('2000-01-01','2000-12-31')
.filter(ee.Filter.lt('CLOUD_COVER', 40))
.sort('CLOUD_COVER')
.median()
s2 = clippedimg(s2)

Map.addLayer(s2, {
  bands: ['SR_B3','SR_B2','SR_B1'],
  min: 7000,
  max: 15000
}, 'True Color_2000')

Map.centerObject(area, 10);
